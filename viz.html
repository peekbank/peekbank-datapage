Interactive visualizations

Explore the time course of children’s word recognition across the datasets in Peekbank. Pick datasets, an age range, and words; the plots update as you go. These are the same analyses as the classic Peekbank Shiny app (following Fernald et al., 2008), computed in your browser from the current database release.

function interval(range = [], options = {}) {
  const [min = 0, max = 1] = range;
  const {
    step = .001,
    label = null,
    value = [min, max],
    format = ([start, end]) => `${start} … ${end}`,
    color,
    width,
    theme,
    __ns__ = randomScope(),
  } = options;

  const css = `
#${__ns__} {
  //font: 13px/1.2 var(--sans-serif);
  display: flex;
  align-items: baseline;
  flex-wrap: wrap;
  max-width: 100%;
  width: auto;
  flex-direction: column;
}
@media only screen and (min-width: 30em) {
  #${__ns__} {
    flex-wrap: nowrap;
    width: 360px;
  }
}
#${__ns__} .label {
  //width: 120px;
  //padding: 5px 0 4px 0;
  //padding: 5px 0 0 0;
  //margin-right: 6.5px;
  flex-shrink: 0;
}
#${__ns__} .form {
  display: flex;
  width: 100%;
}
#${__ns__} .range {
  flex-shrink: 1;
  width: 100%;
}
#${__ns__} .range-slider {
  width: 100%;
  margin-bottom: .3em;
  margin-top: .3em;
}
  `;
  
  const $range = rangeInput({min, max, value: [value[0], value[1]], step, color, width, theme});
  const $output = ihtml`<output>`;
  const $view = ihtml`<div id=${__ns__}>
${label == null ? '' : ihtml`<div class="label">${label}`}
<div class=form>
  <div class=range>
    ${$range}<div class=range-output>${$output}</div>
  </div>
</div>
${ihtml`<style>${css}`}
  `;

  const update = () => {
    const content = format([$range.value[0], $range.value[1]]);
    if(typeof content === 'string') $output.value = content;
    else {
      while($output.lastChild) $output.lastChild.remove();
      $output.appendChild(content);
    }
  };
  $range.oninput = update;
  update();
  
  return Object.defineProperty($view, 'value', {
    get: () => $range.value,
    set: ([a, b]) => {
      $range.value = [a, b];
      update();
    },
  });
}

function rangeInput(options = {}) {
  const {
    min = 0,
    max = 100,
    step = 'any',
    value: defaultValue = [min, max],
    color,
    width,
    theme = theme_Flat,
  } = options;
  
  const controls = {};
  const scope = randomScope();
  const clamp = (a, b, v) => v < a ? a : v > b ? b : v;

  // Will be used to sanitize values while avoiding floating point issues.
  const input = ihtml`<input type=range ${{min, max, step}}>`;
  
  const dom = ihtml`<div class=${`${scope} range-slider`} style=${{
    color,
    width: cssLength(width),
  }}>
  ${controls.track = ihtml`<div class="range-track">
    ${controls.zone = ihtml`<div class="range-track-zone">
      ${controls.range = ihtml`<div class="range-select" tabindex=0>
        ${controls.min = ihtml`<div class="thumb thumb-min" tabindex=0>`}
        ${controls.max = ihtml`<div class="thumb thumb-max" tabindex=0>`}
      `}
    `}
  `}
  ${ihtml`<style>${theme.replace(/:scope\b/g, '.'+scope)}`}
</div>`;

  let value = [], changed = false;
  Object.defineProperty(dom, 'value', {
    get: () => [...value],
    set: ([a, b]) => {
      value = sanitize(a, b);
      updateRange();
    },
  });

  const sanitize = (a, b) => {
    a = isNaN(a) ? min : ((input.value = a), input.valueAsNumber);
    b = isNaN(b) ? max : ((input.value = b), input.valueAsNumber);
    return [Math.min(a, b), Math.max(a, b)];
  }
  
  const updateRange = () => {
    const ratio = v => (v - min) / (max - min);
    dom.style.setProperty('--range-min', `${ratio(value[0]) * 100}%`);
    dom.style.setProperty('--range-max', `${ratio(value[1]) * 100}%`);
  };

  const dispatch = name => {
    dom.dispatchEvent(new Event(name, {bubbles: true}));
  };
  const setValue = (vmin, vmax) => {
    const [pmin, pmax] = value;
    value = sanitize(vmin, vmax);
    updateRange();
    // Only dispatch if values have changed.
    if(pmin === value[0] && pmax === value[1]) return;
    dispatch('input');
    changed = true;
  };
  
  setValue(...defaultValue);
  
  // Mousemove handlers.
  const handlers = new Map([
    [controls.min, (dt, ov) => {
      const v = clamp(min, ov[1], ov[0] + dt * (max - min));
      setValue(v, ov[1]);
    }],
    [controls.max, (dt, ov) => {
      const v = clamp(ov[0], max, ov[1] + dt * (max - min));
      setValue(ov[0], v);
    }],
    [controls.range, (dt, ov) => {
      const d = ov[1] - ov[0];
      const v = clamp(min, max - d, ov[0] + dt * (max - min));
      setValue(v, v + d);
    }],
  ]);
  
  // Returns client offset object.
  const pointer = e => e.touches ? e.touches[0] : e;
  // Note: Chrome defaults "passive" for touch events to true.
  const on  = (e, fn) => e.split(' ').map(e => document.addEventListener(e, fn, {passive: false}));
  const off = (e, fn) => e.split(' ').map(e => document.removeEventListener(e, fn, {passive: false}));
  
  let initialX, initialV, target, dragging = false;
  function handleDrag(e) {
    // Gracefully handle exit and reentry of the viewport.
    if(!e.buttons && !e.touches) {
      handleDragStop();
      return;
    }
    dragging = true;
    const w = controls.zone.getBoundingClientRect().width;
    e.preventDefault();
    handlers.get(target)((pointer(e).clientX - initialX) / w, initialV);
  }
  
  
  function handleDragStop(e) {
    off('mousemove touchmove', handleDrag);
    off('mouseup touchend', handleDragStop);
    if(changed) dispatch('change');
  }
  
  invalidation.then(handleDragStop);
  
  dom.ontouchstart = dom.onmousedown = e => {
    dragging = false;
    changed = false;
    if(!handlers.has(e.target)) return;
    on('mousemove touchmove', handleDrag);
    on('mouseup touchend', handleDragStop);
    e.preventDefault();
    e.stopPropagation();
    
    target = e.target;
    initialX = pointer(e).clientX;
    initialV = value.slice();
  };
  
  controls.track.onclick = e => {
    if(dragging) return;
    changed = false;
    const r = controls.zone.getBoundingClientRect();
    const t = clamp(0, 1, (pointer(e).clientX - r.left) / r.width);
    const v = min + t * (max - min);
    const [vmin, vmax] = value, d = vmax - vmin;
    if(v < vmin) setValue(v, v + d);
    else if(v > vmax) setValue(v - d, v);
    if(changed) dispatch('change');
  };
  
  return dom;
}

function randomScope(prefix = 'scope-') {
  return prefix + (performance.now() + Math.random()).toString(32).replace('.', '-');
}

function lgamma(z) {
  const g = [676.5203681218851, -1259.1392167224028, 771.32342877765313,
             -176.61502916214059, 12.507343278686905, -0.13857109526572012,
             9.9843695780195716e-6, 1.5056327351493116e-7];
  if (z < 0.5) return Math.log(Math.PI / Math.sin(Math.PI * z)) - lgamma(1 - z);
  z -= 1;
  let x = 0.99999999999980993;
  for (let i = 0; i < g.length; i++) x += g[i] / (z + i + 1);
  const t = z + g.length - 0.5;
  return 0.5 * Math.log(2 * Math.PI) + (z + 0.5) * Math.log(t) - t + Math.log(x);
}

function betacf(x, a, b) {
  const MAXIT = 200, EPS = 3e-12, FPMIN = 1e-300;
  const qab = a + b, qap = a + 1, qam = a - 1;
  let c = 1, d = 1 - qab * x / qap;
  if (Math.abs(d) < FPMIN) d = FPMIN;
  d = 1 / d;
  let h = d;
  for (let m = 1; m <= MAXIT; m++) {
    const m2 = 2 * m;
    let aa = m * (b - m) * x / ((qam + m2) * (a + m2));
    d = 1 + aa * d; if (Math.abs(d) < FPMIN) d = FPMIN;
    c = 1 + aa / c; if (Math.abs(c) < FPMIN) c = FPMIN;
    d = 1 / d; h *= d * c;
    aa = -(a + m) * (qab + m) * x / ((a + m2) * (qap + m2));
    d = 1 + aa * d; if (Math.abs(d) < FPMIN) d = FPMIN;
    c = 1 + aa / c; if (Math.abs(c) < FPMIN) c = FPMIN;
    d = 1 / d;
    const del = d * c;
    h *= del;
    if (Math.abs(del - 1) < EPS) break;
  }
  return h;
}

function incBeta(x, a, b) {
  if (x <= 0) return 0;
  if (x >= 1) return 1;
  const bt = Math.exp(lgamma(a + b) - lgamma(a) - lgamma(b) +
                      a * Math.log(x) + b * Math.log(1 - x));
  return x < (a + 1) / (a + b + 2)
    ? bt * betacf(x, a, b) / a
    : 1 - bt * betacf(1 - x, b, a) / b;
}

function betaQuantile(p, a, b) {
  let lo = 0, hi = 1;
  for (let i = 0; i < 80; i++) {
    const mid = (lo + hi) / 2;
    if (incBeta(mid, a, b) < p) lo = mid; else hi = mid;
  }
  return (lo + hi) / 2;
}

// 95% credible interval for x successes of n (Jeffreys prior), as in the app
function bayesCI(x, n) {
  if (n === 0) return [NaN, NaN];
  return [betaQuantile(0.025, x + 0.5, n - x + 0.5),
          betaQuantile(0.975, x + 0.5, n - x + 0.5)];
}
STEP_MS = 25

function trialRT(t0, runs) {
  // keep only samples at t >= 0, requiring a sample at exactly t = 0
  if (t0 > 0) return null; // no data at 0
  let skip = (0 - t0) / STEP_MS; // samples to drop before t = 0
  const post = [];
  for (const [aoi, len] of runs) {
    if (skip >= len) { skip -= len; continue; }
    post.push([aoi, len - skip]);
    skip = 0;
  }
  if (post.length === 0) return null; // no data at 0
  const onset = post[0][0];
  if (onset !== 0 && onset !== 1) return ({ rt: null, shift_type: "other" });
  let landingIdx = -1;
  for (let i = 1; i < post.length; i++) {
    const v = post[i][0];
    if (v !== onset && (v === 0 || v === 1)) { landingIdx = i; break; }
  }
  if (landingIdx === -1) return ({ rt: null, shift_type: "no shift" });
  const landing = post[landingIdx][0];
  const shift_type =
    onset === 1 && landing === 0 ? "D-T" :
    onset === 0 && landing === 1 ? "T-D" : "other";
  let before = 0;
  for (let i = 0; i < landingIdx; i++) before += post[i][1];
  return ({
    rt: (before + 1) * STEP_MS,           // landing-time RT
    shift_start_rt: (post[0][1] + 1) * STEP_MS,
    shift_type,
  });
}
function makeAgeBinner(ages, nbins) {
  if (nbins <= 1 || ages.length === 0) {
    return ({ bin: () => 0, labels: ["all ages"] });
  }
  const lo = Math.min(...ages), hi = Math.max(...ages);
  // R's cut() widens the range by 0.1% to include the endpoints
  const pad = (hi - lo || 1) * 0.001;
  const start = lo - pad, end = hi + pad;
  const w = (end - start) / nbins;
  const fmt = (x) => +x.toFixed(1);
  const labels = Array.from({ length: nbins }, (_, i) =>
    `(${fmt(start + i * w)},${fmt(start + (i + 1) * w)}]`);
  return ({
    // right-closed (lo, hi] intervals, matching R's cut()
    bin: (a) => Math.min(nbins - 1, Math.max(0, Math.ceil((a - start) / w) - 1)),
    labels,
  });
}

joined = {
  const useAll = sel_words.includes("All");
  const wordSet = new Set(sel_words);
  const [ageLo, ageHi] = age_range;
  const rows = []; // one row per run-group (trial administration)
  for (const s of slices) {
    const adminById = new Map(s.admins.map((a) => [a[0], a]));
    const ttById = new Map(s.trial_types.map((t) => [t[0], t]));
    const trialById = new Map(s.trials.map((t) => [t[0], t]));
    for (const [aid, tid, t0, runs] of s.runs) {
      const admin = adminById.get(aid);

        age: admin[1], subject: admin[2],
        word: useAll ? "All" : word, trial_order: trial[1],

accuracyData = {
  const [wLo, wHi] = analysis_window;
  const perTrial = [];
  for (const r of groupedRows) {
    let t = r.t0, n = 0, p = 0;
    for (const [aoi, len] of r.runs) {
      for (let k = 0; k < len; k++, t += STEP_MS) {
        if (t < wLo || t > wHi || aoi > 1) continue;
        n += 1;
        if (aoi === 0) p += 1;
      }
    }
    if (n > 0) perTrial.push({ age_bin: r.age_bin, word: r.word, prop: p / n });
  }
  const groups = d3.group(perTrial, (d) => d.age_bin, (d) => d.word);
  const out = [];
  for (const [bin, byWord] of groups) {
    for (const [word, rows] of byWord) {
      const mean = d3.mean(rows, (d) => d.prop);
      const sem = (d3.deviation(rows, (d) => d.prop) || 0) / Math.sqrt(rows.length);
      out.push({
        age_bin: bin, word, n_trials: rows.length, mean,
        ci_lower: mean - 1.96 * sem, ci_upper: mean + 1.96 * sem,
      });
    }
  }
  return out.sort((a, b) => d3.ascending(a.age_bin, b.age_bin));
}

rtMeans = {
  const dt = rtData.filter((d) => d.shift_type === "D-T");
  const out = [];
  for (const [bin, byWord] of d3.group(dt, (d) => d.age_bin, (d) => d.word)) {
    for (const [word, rows] of byWord) {
      const mean = d3.mean(rows, (d) => d.rt);
      const sem = (d3.deviation(rows, (d) => d.rt) || 0) / Math.sqrt(rows.length);
      out.push({
        age_bin: bin, word, n_trials: rows.length, mean_rt: mean,
        ci_lower: mean - 1.96 * sem, ci_upper: mean + 1.96 * sem,
      });
    }
  }
  return out.sort((a, b) => d3.ascending(a.age_bin, b.age_bin));
}

Onset-contingent looking

Proportion of trials on which children have shifted away from the image they were fixating at word onset, split by whether they started on the distractor (D-T, solid lines) or on the target (T-D, dashed lines).

  • Data
onsetData = {
  const [tLo, tHi] = plot_window;
  const shiftByTrial = new Map();
  for (const r of groupedRows) {
    const res = trialRT(r.t0, r.runs);
    if (res && (res.shift_type === "D-T" || res.shift_type === "T-D")) {
      shiftByTrial.set(`${r.aid}|${r.tid}`, res.shift_type);
    }
  }
  const acc = new Map();
  for (const r of groupedRows) {
    const st = shiftByTrial.get(`${r.aid}|${r.tid}`);
    if (!st) continue;
    let t = r.t0;
    for (const [aoi, len] of r.runs) {
      for (let k = 0; k < len; k++, t += STEP_MS) {
        if (t < 0 || t >= tHi) continue;
        const key = `${t}|${st}|${r.age_bin}|${r.word}`;
        let e = acc.get(key);
        if (!e) acc.set(key, e = {
          t, shift_type: st, age_bin: r.age_bin, word: r.word, n: 0, s: 0,
        });
        e.n += 1;
        // proportion looking at the *other* image than the onset image
        if (st === "T-D" ? aoi === 1 : aoi === 0) e.s += 1;
      }
    }
  }
  return Array.from(acc.values(), (e) => ({
    t: e.t, shift_type: e.shift_type, age_bin: e.age_bin, word: e.word,
    n: e.n, prop_shifted: e.s / e.n,
  })).sort((a, b) => a.t - b.t);
}
{
  const facets = new Set(onsetData.map((d) => d.age_bin));
  return Plot.plot({
    width, height: Math.max(360, 240 * Math.ceil(facets.size / 2)),
    grid: true,

    y: { label: "Prop. shifting from onset image", domain: [0, 1] },
    color: { legend: true, label: "Word" },

    marks: [

        x: "t", y: "prop_shifted", stroke: "word",

      Plot.ruleX([0], { strokeDasharray: "4 4" }),
    ],
  });
}
