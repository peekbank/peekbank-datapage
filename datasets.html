Datasets

Every dataset in the current Peekbank release. Search by name, citation, language, or method; expand a dataset for its full citation and details. When you use a dataset, cite its original publication as well as the database (see how to cite).

manifest = d3.json((typeof window !== "undefined" && window.PEEKBANK_DATA_BASE || "slices/") + "manifest.json")
viewof ds_query = Inputs.search(manifest.datasets, {
  placeholder: "Search datasets…",
  columns: ["name", "shortcite", "cite", "languages", "methods",
            "native_languages"],
})
html`<p style="color:#6b7280">${ds_query.length} of
${manifest.datasets.length} datasets · release ${manifest.version} ·
${d3.sum(ds_query, (d) => d.n_subjects).toLocaleString()} children ·
${d3.sum(ds_query, (d) => d.n_admins).toLocaleString()} sessions ·
${d3.sum(ds_query, (d) => d.n_trials).toLocaleString()} trials</p>`
html`<div>${ds_query.map((d) => html`
<details class="pb-dataset-item">
  <summary>${d.name}
    <span style="font-weight:400; color:#6b7280"> — ${d.shortcite ?? ""}</span>
  </summary>
  <div class="pb-ds-body">
    <p>${d.cite ?? d.shortcite ?? "No citation recorded."}</p>
    <p class="pb-ds-meta">
      Explore this dataset in the
      <a href="accuracy.html">Accuracy</a> ·
      <a href="rt.html">Reaction Time</a>${d.longitudinal
        ? html` · <a href="longitudinal.html">Longitudinal</a>` : ""}${
        d.n_subjects_with_cdi > 0
        ? html` · <a href="cdi.html">CDI</a>` : ""} explorers,
      or in the
      <a href="https://stanford.redivis.com/datasets/a3v0-3drqa89kg">peekbank
      dataset on Redivis</a>.
    </p>
  </div>
  <div class="pb-ds-chips">
    <span class="pb-chip">${d.n_subjects.toLocaleString()} children</span>
    <span class="pb-chip">${d.n_admins.toLocaleString()} sessions</span>
    <span class="pb-chip">${d.n_trials.toLocaleString()} trials</span>
    <span class="pb-chip">${d.age_min}–${d.age_max} mo</span>
    <span class="pb-chip">${d.n_words} words</span>
    ${(d.methods ?? []).map((m) => html`<span class="pb-chip">${m.replace("_", " ")}</span>`)}
    ${(d.languages ?? []).map((l) => html`<span class="pb-chip">${l}</span>`)}
    ${d.longitudinal ? html`<span class="pb-chip" style="background:#dbeafe">longitudinal</span>` : ""}
    ${d.n_subjects_with_cdi > 0
      ? html`<span class="pb-chip" style="background:#dcfce7">CDI: ${d.n_subjects_with_cdi}</span>` : ""}
  </div>
</details>`)}</div>`
